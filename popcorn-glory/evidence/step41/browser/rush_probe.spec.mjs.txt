// Temporary (Step 41 measurement, deleted after): the art41 rush on any build, its render cost.
import { test, expect } from '../lib/pop.mjs';
async function popOnce(pop) {
  const s = await pop.state();
  if (!s.run.heating && Object.keys(s.run.glass).length === 0 && s.run.cooldown_left <= 0) await pop.hold('ring', Math.round(s.run.window[0] + (s.run.window[1] - s.run.window[0]) * 0.25));
}
test('rush render cost', async ({ pop }, testInfo) => {
  test.skip(testInfo.project.name !== 'p720', 'p720 only');
  test.setTimeout(300_000);
  await pop.open('./#factory');
  await pop.page.waitForTimeout(1500);
  const fresh = await pop.call('dev_render_info');
  for (let i = 0; i < 4; i++) { await popOnce(pop); await pop.call('dev_advance', 9000); }
  await expect.poll(async () => (await pop.state()).run.open.sign, { timeout: 25_000 }).toBe(true);
  await pop.call('dev_money', 400000);
  await pop.tap('floor_top');
  for (let t = 0; t < 8; t++) await pop.tap('upgrade');
  await pop.tap('badge_top');
  while ((await pop.state()).run.now_ms < 30 * 60000) { await popOnce(pop); await pop.call('dev_advance', 60000); }
  const rows = [];
  for (let t = 0; t < 12; t++) {
    await popOnce(pop); await pop.call('dev_advance', 1500); await pop.page.waitForTimeout(400);
    const v = (await pop.state()).view;
    const info = await pop.call('dev_render_info');
    rows.push({ people: v.customers + v.leaving + v.passers, dc: info.draw_calls, obj: info.objects, fps: info.fps });
  }
  testInfo.annotations.push({ type: 'rush', description: JSON.stringify({ fresh: { dc: fresh.draw_calls, obj: fresh.objects, fps: fresh.fps }, rows }) });
});

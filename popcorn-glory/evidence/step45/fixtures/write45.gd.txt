extends SceneTree
func _init() -> void:
	var out := OS.get_environment("OUT")
	var T = load("res://tests/unit/test_factory_rules.gd").new()
	var f = T._forty(44, 81)
	f.advance(300000)
	var t1 := PopSaveDoc.encode("factory", "0.45.0", 300000, int(Time.get_unix_time_from_system()), f.to_save())
	var s: PopRules = PopRegistry.make_rules("stand", null)
	s.money = 777
	s.level = 3
	s.earned = 900
	var t2 := PopSaveDoc.encode("stand", "0.45.0", 60000, int(Time.get_unix_time_from_system()), s.to_save())
	for pair in [["factory_0.45.json", t1], ["stand_0.45.json", t2]]:
		var fa := FileAccess.open(out + "/" + pair[0], FileAccess.WRITE)
		fa.store_string(pair[1])
		fa.close()
	print("wrote factory %d bytes (machines %d, money %d), stand %d bytes" % [t1.length(), f.machines.size(), f.money, t2.length()])
	quit(0)
